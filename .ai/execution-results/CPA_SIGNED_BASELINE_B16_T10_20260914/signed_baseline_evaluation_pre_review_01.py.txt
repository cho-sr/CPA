"""Offline only: finalized two-arm sources/images against private sender truth."""
from __future__ import annotations

import json
from pathlib import Path
import time

import torch
from torch.nn import functional as F

from .full_update_evaluation import evaluate_candidate
from .report_full_update_iterations import _make_lpips_metric
from .signed_baseline_observation import digest, read_bundle
from .signed_baseline import write_json, quantization_fit


def _completed(folder):
    folder = Path(folder)
    result = json.loads((folder/"result.json").read_text(encoding="utf-8"))
    if result["status"] != "COMPLETED" or result["cpa_iterations"] != 25000 or result["fia_iterations"] != 25000:
        raise ValueError("all final iterations must be complete before evaluation")
    for filename, key in (("candidate.pt", "candidate_sha256"), ("sources.pt", "sources_sha256")):
        if digest(folder/filename) != result[key]:
            raise ValueError("final artifact changed before offline evaluation")
    return result


def embedding_evaluation(target, truth_z):
    from scipy.optimize import linear_sum_assignment
    if target.shape != truth_z.shape or not bool(torch.isfinite(target).all() & torch.isfinite(truth_z).all()):
        raise ValueError("invalid source/embedding shape or values")
    cosine = F.normalize(target.double(), dim=1) @ F.normalize(truth_z.double(), dim=1).T
    rows, cols = linear_sum_assignment(-cosine.numpy())
    pairs = [{"source_index": int(i), "truth_index": int(j), "cosine": float(cosine[i,j])}
             for i,j in zip(rows, cols)]
    return {"matching": "Hungarian_max_embedding_cosine_offline_only", "per_source": pairs,
            "mean_cosine": sum(p["cosine"] for p in pairs)/len(pairs),
            "all_sources_included": True, "adapter": "abs(S_hat); no scaling/permutation before FIA"}


@torch.no_grad()
def evaluate(*, naive, reference, private_observation, public_observation, output):
    """No attack/optimizer invocation; original labels are never consumed here."""
    torch.set_num_threads(8)
    folders = {"reference": Path(reference), "naive": Path(naive)}
    output = Path(output)
    if output.exists():
        raise FileExistsError(output)
    # Validate both arms and hashes before opening any private evaluation data.
    final_results = {arm: _completed(folder) for arm, folder in folders.items()}
    inits = {arm: json.loads((folder/"fia_initial.json").read_text(encoding="utf-8")) for arm, folder in folders.items()}
    if inits["reference"]["tensor_sha256"] != inits["naive"]["tensor_sha256"]:
        raise ValueError("FIA pixel initialization differs across arms")
    for field in ("cudnn_benchmark", "cudnn_deterministic", "deterministic_algorithms", "deterministic_warn_only",
                  "matmul_tf32", "cudnn_tf32", "cublas_workspace_config", "threads"):
        if inits["reference"]["runtime"][field] != inits["naive"]["runtime"][field]:
            raise ValueError(f"arm numerical runtime differs: {field}")
    truth_path = Path(private_observation)/"evaluation_truth.pt"
    sender = json.loads((Path(private_observation)/"sender_validation.json").read_text(encoding="utf-8"))
    if digest(truth_path) != sender["evaluation_truth_sha256"]:
        raise ValueError("collector truth changed")
    truth = torch.load(truth_path, map_location="cpu", weights_only=True)
    output.mkdir(parents=True)
    tick = time.perf_counter()
    mean = truth["x"].new_tensor((.485,.456,.406)).reshape(1,3,1,1)
    std = truth["x"].new_tensor((.229,.224,.225)).reshape(1,3,1,1)
    originals = (truth["x"]*std+mean).clamp(0,1)
    metric, provenance = _make_lpips_metric()
    arm_data, candidates = {}, {}
    for arm, folder in folders.items():
        candidate = torch.load(folder/"candidate.pt", map_location="cpu", weights_only=True)
        if set(candidate) != {"pixels"}:
            raise ValueError("label-free final candidate required")
        pixels = candidate["pixels"]
        sources = torch.load(folder/"sources.pt", map_location="cpu", weights_only=True)
        image = evaluate_candidate(folder/"candidate.pt", truth_path, include_labels=False)
        lpips = []
        for pair in image["per_image"]:
            i,j = pair["candidate_index"],pair["truth_index"]
            metric.reset()
            value = metric((pixels[i:i+1].float()-.5)*2, (originals[j:j+1].float()-.5)*2)
            if value.numel()!=1 or not bool(torch.isfinite(value)):
                raise ValueError("nonfinite LPIPS")
            pair["lpips"] = float(value)
            lpips.append(float(value))
        image["lpips"] = "VGG_same_pixel_MSE_Hungarian_pairs"
        image["mean_lpips"] = sum(lpips)/len(lpips)
        embedding = embedding_evaluation(sources["target_z"], truth["z"])
        arm_data[arm] = {"image_oracle": image, "embedding_oracle": embedding,
                         "final_result": final_results[arm], "initialization": inits[arm]}
        candidates[arm] = pixels
        print(f"offline {arm}: PSNR={image['mean_psnr_db']:.6f}, SSIM={image['mean_ssim']:.6f}, LPIPS={image['mean_lpips']:.6f}", flush=True)
    paired = []
    for j in range(16):
        pairs = {arm: next(p for p in data["image_oracle"]["per_image"] if p["truth_index"]==j)
                 for arm,data in arm_data.items()}
        paired.append({"truth_index": j, "naive_candidate_index": pairs["naive"]["candidate_index"],
                       "reference_candidate_index": pairs["reference"]["candidate_index"],
                       **{key: pairs["naive"][key]-pairs["reference"][key] for key in ("psnr_db","ssim","lpips","mse")}})
    deltas = {key: arm_data["naive"]["image_oracle"][key]-arm_data["reference"]["image_oracle"][key]
              for key in ("mean_psnr_db","mean_ssim","mean_lpips","mean_mse")}
    raw_path = Path(private_observation)/"raw_updates.pt"
    if digest(raw_path) != sender["raw_updates_sha256"]:
        raise ValueError("raw update file changed")
    raw = torch.load(raw_path, map_location="cpu", weights_only=True)["updates"]
    manifest, observed = read_bundle(Path(public_observation))
    update_errors = {}
    # Raw Delta comparison remains evaluation-only; no error statistics enter the public bundle.
    for name,G in raw.items():
        q,s = observed[name]
        sumsq = normsq = 0.
        maxabs = 0.
        a,b = G.flatten(), q.flatten()
        for start in range(0,G.numel(),262144):
            orig = a[start:start+262144].double()
            decoded = (b[start:start+262144].float()*s).double()
            error = decoded-orig
            sumsq += float(error.square().sum())
            normsq += float(orig.square().sum())
            maxabs = max(maxabs,float(error.abs().max()))
        update_errors[name] = {"mse": sumsq/G.numel(), "relative_l2": (sumsq/normsq)**.5 if normsq else None,
                               "relative_l2_null_reason": "zero_original_norm" if not normsq else None,
                               "max_abs_error": maxabs, "sender_raw_fit": quantization_fit(G,q,s)}
    _figures(output, originals, candidates, arm_data)
    # No final candidate was modified or rematched by the embedding evaluation.
    for arm,folder in folders.items():
        assert _completed(folder) == final_results[arm]
    comparison = {"status": "COMPLETED", "arms": arm_data, "naive_minus_reference": deltas,
                  "paired_naive_minus_reference_by_truth": paired,
                  "embedding_mean_cosine_delta": arm_data["naive"]["embedding_oracle"]["mean_cosine"]-arm_data["reference"]["embedding_oracle"]["mean_cosine"],
                  "update_errors_evaluation_only": update_errors, "lpips_provenance": provenance,
                  "truth_sha256": digest(truth_path), "communication": manifest["communication"],
                  "wall_seconds": time.perf_counter()-tick, "batch_count": 1, "seed": 42,
                  "success_threshold": None, "generalization": "NOT_ESTABLISHED_SINGLE_BATCH_SINGLE_SEED",
                  "candidate_hashes_unchanged": True, "fia_initializations_equal": True}
    write_json(output/"comparison.json", comparison)
    lines = ["# FP32 Reference / signed INT4 naive — B16/T10", "",
             "[FACT] One new local update; frozen Conv, all FC trained. Both arms: raw Delta/raw sq → official CPA 25,000 → abs sources → Direct FIA 25,000.",
             "", "| Arm | PSNR dB | SSIM | VGG LPIPS | Embedding cosine |", "|---|---:|---:|---:|---:|"]
    for arm,data in arm_data.items():
        e = data["image_oracle"]
        lines.append(f"| {arm} | {e['mean_psnr_db']:.6f} | {e['mean_ssim']:.6f} | {e['mean_lpips']:.6f} | {data['embedding_oracle']['mean_cosine']:.6f} |")
    lines += ["", "[FACT] Naive minus reference: " + json.dumps(deltas), "",
              "[FACT] All 16 final candidates evaluated. One pixel-MSE Hungarian matching per arm supplies PSNR/SSIM/LPIPS pairs. Embedding matching is separate and offline only.",
              "[FACT] Identical FIA initialization hashes and numerical runtime settings. No true-label attack, oracle selection, reranking, or quantization-aware objective.",
              "[UNKNOWN] No success threshold, significance, or generalization established by this single batch/seed.",
              "[FACT] all16_comparison.png and all16_compact.png are evaluation-only figures."]
    (output/"report.md").write_text("\n".join(lines)+"\n",encoding="utf-8")
    return comparison


def _figures(output, originals, candidates, arm_data):
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    for compact in (False,True):
        fig, axes = plt.subplots(4 if compact else 16, 12 if compact else 3,
                                 figsize=(24,8) if compact else (8,38))
        for j in range(16):
            row,offset = (j//4,(j%4)*3) if compact else (j,0)
            axes[row,offset].imshow(originals[j].permute(1,2,0))
            axes[row,offset].set_title(f"{j+1} Original (offline)",fontsize=8)
            for col,arm in enumerate(("reference","naive"),1):
                pair = next(p for p in arm_data[arm]["image_oracle"]["per_image"] if p["truth_index"]==j)
                axes[row,offset+col].imshow(candidates[arm][pair["candidate_index"]].permute(1,2,0))
                axes[row,offset+col].set_title("FP32 Reference" if arm=="reference" else "INT4 naive",fontsize=8)
            for col in range(3):
                axes[row,offset+col].axis("off")
        fig.tight_layout(pad=.5)
        fig.savefig(output/("all16_compact.png" if compact else "all16_comparison.png"),dpi=110)
        plt.close(fig)
